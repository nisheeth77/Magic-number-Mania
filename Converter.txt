import re
import sys
from pathlib import Path


# ============================================================
# COMMON HELPERS
# ============================================================

def read_file(file_path):
    path = Path(file_path)

    if not path.exists():
        raise FileNotFoundError(f"File not found: {path}")

    return path.read_text(encoding="utf-8")


# ============================================================
# MDD VALIDATION
# ============================================================

def extract_mdd_questions(mdd_content):
    """
    Extract question IDs declared in MDD.

    Example:
        Q1 "Do you own a car?"
        Q2 "Employment status"

    Returns:
        {"Q1", "Q2"}
    """

    question_ids = set()

    for line in mdd_content.splitlines():

        line = line.strip()

        # Ignore comments / headers
        if (
            not line
            or line.startswith("'")
            or line == "Metadata"
            or line.startswith("Version")
            or line.startswith("{")
            or line.startswith("}")
        ):
            continue

        match = re.match(
            r'^([A-Za-z_][A-Za-z0-9_]*)\s+"',
            line
        )

        if match:
            question_ids.add(match.group(1))

    return question_ids


def validate_mdd(mdd_file):
    """
    Validate the generated MDD structure.
    """

    errors = []

    try:
        content = read_file(mdd_file)

    except Exception as error:
        return False, [str(error)], set()

    lines = content.splitlines()

    # --------------------------------------------------------
    # Check Metadata header
    # --------------------------------------------------------

    if "Metadata" not in content:
        errors.append(
            "MDD must contain 'Metadata'."
        )

    # --------------------------------------------------------
    # Check Version
    # --------------------------------------------------------

    if not re.search(
        r"\bVersion\s*\(\s*\d+\s*\)",
        content
    ):
        errors.append(
            "MDD Metadata Version is missing."
        )

    # --------------------------------------------------------
    # Check brackets/braces
    # --------------------------------------------------------

    if content.count("{") != content.count("}"):
        errors.append(
            "MDD has unbalanced { } braces."
        )

    if content.count("[") != content.count("]"):
        errors.append(
            "MDD has unbalanced [ ] brackets."
        )

    # --------------------------------------------------------
    # Extract question IDs
    # --------------------------------------------------------

    questions = extract_mdd_questions(content)

    if not questions:
        errors.append(
            "No questions were found in the MDD."
        )

    # --------------------------------------------------------
    # Duplicate question detection
    # --------------------------------------------------------

    found_questions = []

    for line in lines:

        line = line.strip()

        match = re.match(
            r'^([A-Za-z_][A-Za-z0-9_]*)\s+"',
            line
        )

        if match:
            found_questions.append(
                match.group(1)
            )

    duplicates = {
        qid
        for qid in found_questions
        if found_questions.count(qid) > 1
    }

    if duplicates:
        errors.append(
            "Duplicate MDD question IDs: "
            + ", ".join(sorted(duplicates))
        )

    # --------------------------------------------------------
    # Validate categorical questions
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        stripped = line.strip()

        if stripped.startswith("Categorical"):

            if not re.search(
                r"\[\s*\d+\s*\.\.\s*(\d+|\*)\s*\]",
                stripped
            ):
                errors.append(
                    f"Line {line_number}: "
                    "Invalid Categorical cardinality."
                )

    # --------------------------------------------------------
    # Validate INDEX syntax
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        if "INDEX(" in line:

            if not re.search(
                r"INDEX\(\d+\)",
                line
            ):
                errors.append(
                    f"Line {line_number}: "
                    "Invalid INDEX syntax."
                )

    return (
        len(errors) == 0,
        errors,
        questions
    )


# ============================================================
# MRS VALIDATION
# ============================================================

def validate_mrs(mrs_file, mdd_questions):
    """
    Validate the generated mScriptBasic-style MRS file.
    """

    errors = []

    try:
        content = read_file(mrs_file)

    except Exception as error:
        return False, [str(error)]

    lines = content.splitlines()

    # --------------------------------------------------------
    # Check Sub Routing
    # --------------------------------------------------------

    if not re.search(
        r"\bSub\s+Routing\s*\(\s*\)",
        content,
        re.IGNORECASE
    ):
        errors.append(
            "MRS is missing 'Sub Routing()'."
        )

    # --------------------------------------------------------
    # Check End Sub
    # --------------------------------------------------------

    if not re.search(
        r"\bEnd\s+Sub\b",
        content,
        re.IGNORECASE
    ):
        errors.append(
            "MRS is missing 'End Sub'."
        )

    # --------------------------------------------------------
    # Check IF / END IF balance
    # --------------------------------------------------------

    if_count = 0
    endif_count = 0

    for line in lines:

        stripped = line.strip()

        if re.match(
            r"^(If|ElseIf)\b",
            stripped,
            re.IGNORECASE
        ):

            # ElseIf does NOT open another IF block
            if stripped.lower().startswith("if "):
                if_count += 1

        if re.match(
            r"^End\s+If\b",
            stripped,
            re.IGNORECASE
        ):
            endif_count += 1

    if if_count != endif_count:

        errors.append(
            f"Unbalanced If blocks: "
            f"{if_count} If statements but "
            f"{endif_count} End If statements."
        )

    # --------------------------------------------------------
    # Validate IF syntax
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        stripped = line.strip()

        if re.match(
            r"^(If|ElseIf)\b",
            stripped,
            re.IGNORECASE
        ):

            if not re.search(
                r"\bThen\s*$",
                stripped,
                re.IGNORECASE
            ):
                errors.append(
                    f"Line {line_number}: "
                    "If/ElseIf statement must end with 'Then'."
                )

    # --------------------------------------------------------
    # Validate ASK statements
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        stripped = line.strip()

        match = re.match(
            r"^Ask\s+([A-Za-z_][A-Za-z0-9_]*)$",
            stripped,
            re.IGNORECASE
        )

        if match:

            question_id = match.group(1)

            if question_id not in mdd_questions:

                errors.append(
                    f"Line {line_number}: "
                    f"Ask references unknown question "
                    f"'{question_id}'."
                )

    # --------------------------------------------------------
    # Validate SetValue question references
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        stripped = line.strip()

        match = re.match(
            r"^([A-Za-z_][A-Za-z0-9_]*)"
            r"\.SetValue\(\{([^}]+)\}\)$",
            stripped,
            re.IGNORECASE
        )

        if match:

            question_id = match.group(1)

            if question_id not in mdd_questions:

                errors.append(
                    f"Line {line_number}: "
                    f"SetValue references unknown question "
                    f"'{question_id}'."
                )

    # --------------------------------------------------------
    # Validate question references in conditions
    # --------------------------------------------------------

    for line_number, line in enumerate(
        lines,
        start=1
    ):

        stripped = line.strip()

        if (
            stripped.lower().startswith("if ")
            or stripped.lower().startswith("elseif ")
        ):

            # Find things like:
            # Q1 = {yes}
            # Q2 <> {no}

            references = re.findall(
                r"\b([A-Za-z_][A-Za-z0-9_]*)"
                r"\s*(?:=|<>)\s*\{",
                stripped
            )

            for question_id in references:

                if question_id not in mdd_questions:

                    errors.append(
                        f"Line {line_number}: "
                        f"Condition references unknown question "
                        f"'{question_id}'."
                    )

    return (
        len(errors) == 0,
        errors
    )


# ============================================================
# COMPILE / VALIDATE BOTH FILES
# ============================================================

def compile_survey(mdd_file, mrs_file):

    print("")
    print("=" * 55)
    print("MDD + MRS VALIDATION")
    print("=" * 55)

    # --------------------------------------------------------
    # Validate MDD
    # --------------------------------------------------------

    print("\nCompiling MDD...")

    mdd_valid, mdd_errors, questions = validate_mdd(
        mdd_file
    )

    if mdd_valid:

        print("MDD: PASS")

        print(
            f"Questions found: {len(questions)}"
        )

    else:

        print("MDD: FAIL")

        for error in mdd_errors:
            print(f"  - {error}")

    # --------------------------------------------------------
    # Only validate MRS references if we have MDD questions
    # --------------------------------------------------------

    print("\nCompiling MRS...")

    mrs_valid, mrs_errors = validate_mrs(
        mrs_file,
        questions
    )

    if mrs_valid:

        print("MRS: PASS")

    else:

        print("MRS: FAIL")

        for error in mrs_errors:
            print(f"  - {error}")

    # --------------------------------------------------------
    # Final result
    # --------------------------------------------------------

    print("")
    print("=" * 55)
    print("FINAL RESULT")
    print("=" * 55)

    if mdd_valid and mrs_valid:

        print("COMPILATION SUCCESSFUL")
        print("Both MDD and MRS passed validation.")

        return True

    else:

        print("COMPILATION FAILED")

        if not mdd_valid:
            print("MDD contains errors.")

        if not mrs_valid:
            print("MRS contains errors.")

        return False


# ============================================================
# MAIN
# ============================================================

if __name__ == "__main__":

    if len(sys.argv) != 3:

        print("")
        print("Usage:")
        print(
            "python compiler.py survey.mdd survey.mrs"
        )

        print("")
        print("Example:")
        print(
            "python compiler.py questionnaire_1.mdd "
            "questionnaire_1.mrs"
        )

        sys.exit(1)

    mdd_file = sys.argv[1]
    mrs_file = sys.argv[2]

    success = compile_survey(
        mdd_file,
        mrs_file
    )

    if success:
        sys.exit(0)

    else:
        sys.exit(1)
